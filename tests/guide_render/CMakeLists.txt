cmake_minimum_required(VERSION 3.16)
project(guide_render C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(guide_render render.c ../../main/guide_ui.c ../../assets/guide/guide_art.c ../../main/guide_data.c ../../main/guide_record.c ../../main/guide_model.c)
foreach(size 12 14 16 20)
    target_sources(guide_render PRIVATE ../../assets/fonts/guide_font_${size}.c)
endforeach()
target_include_directories(guide_render PRIVATE ../../main)
target_link_libraries(guide_render lvgl m)
