cmake_minimum_required(VERSION 3.16)
project(orbit_render C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(orbit_render render.c ../../main/orbit_ui.c ../../main/orbit_model.c ../../main/orbit_language.c ../../assets/fonts/orbit_font_16.c ../../assets/fonts/orbit_font_14.c ../../assets/fonts/orbit_font_20.c ../../assets/fonts/orbit_font_12.c ../../assets/fonts/orbit_font_24.c ../../assets/character/orbit_archivist.c ../../assets/character/orbit_celebration.c ../../assets/wechat/orbit_miniapp.c)
target_sources(orbit_render PRIVATE ../../assets/fonts/orbit_font_tc_12.c ../../assets/fonts/orbit_font_tc_14.c ../../assets/fonts/orbit_font_tc_16.c ../../assets/fonts/orbit_font_tc_20.c ../../assets/fonts/orbit_font_tc_24.c)
target_include_directories(orbit_render PRIVATE ../../main)
target_link_libraries(orbit_render lvgl m)
